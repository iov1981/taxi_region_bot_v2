// ============================================================ // ОЦЕНКА РАССТОЯНИЯ И РАСЧЁТ СТОИМОСТИ В MINI APP // ============================================================ // Функция расчёта цены по тарифной сетке бота function calculatePrice(distanceKm, hasChildChair = false) { if (!distanceKm || distanceKm <= 0) return null; let base = 0; const isCity = distanceKm <= 15.0; if (isCity) { if (distanceKm <= 2.0) { base = 200; } else { base = 200 + ((distanceKm - 2.0) * 35); } } else { base = Math.max(distanceKm * 30, 300); } if (hasChildChair) { base += 100; } return { distanceKm: Math.round(distanceKm * 10) / 10, price: Math.round(base) }; } // Приблизительная оценка дистанции по тексту маршрута function estimateApproxDistance(fromText, toText) { if (!fromText || !toText) return null; const cleanFrom = fromText.toLowerCase().trim(); const cleanTo = toText.toLowerCase().trim(); // Города области const towns = ["кузнецк", "заречный", "каменка", "сердобск", "нижний ломов", "никольск", "белинский", "городище", "мокшан"]; const isIntercity = towns.some(t => cleanFrom.includes(t) || cleanTo.includes(t)); if (isIntercity) { return 45.0; // Средняя ориентировочная дистанция для межгорода } // Если обычная поездка по Пензе (в среднем 4-7 км) return 5.5; } // Обновление плашки стоимости на экране function updateEstimateDisplay() { const fromInput = document.getElementById("from"); const toInput = document.getElementById("to"); const estimateCard = document.getElementById("estimate-card"); const estimateDistance = document.getElementById("estimate-distance"); const estimatePrice = document.getElementById("estimate-price"); if (!fromInput || !toInput || !estimateCard) return; const fromVal = fromInput.value.trim(); const toVal = toInput.value.trim(); // Показываем блок, только если оба адреса заполнены (минимум от 3 символов) if (fromVal.length >= 3 && toVal.length >= 3) { // Проверяем, нажата ли кнопка детского кресла const childTag = document.querySelector('.tag[data-comment="Нужно детское кресло"]'); const hasChildChair = childTag ? childTag.classList.contains("active") : false; const dist = estimateApproxDistance(fromVal, toVal); const res = calculatePrice(dist, hasChildChair); if (res) { estimateDistance.textContent = `Дистанция: ~${res.distanceKm} км`; estimatePrice.textContent = `~${res.price} ₽`; estimateCard.style.display = "block"; return; } } estimateCard.style.display = "none"; } // Слушатели событий ввода адресов и клика по тегам document.addEventListener("DOMContentLoaded", () => { const fromInput = document.getElementById("from"); const toInput = document.getElementById("to"); const swapBtn = document.getElementById("swap-route"); if (fromInput) fromInput.addEventListener("input", updateEstimateDisplay); if (toInput) toInput.addEventListener("input", updateEstimateDisplay); if (swapBtn) { swapBtn.addEventListener("click", () => { setTimeout(updateEstimateDisplay, 50); }); } // При клике на любые теги (включая детское кресло) пересчитывать цену document.querySelectorAll(".tag").forEach(tag => { tag.addEventListener("click", () => { setTimeout(updateEstimateDisplay, 50); }); }); });
